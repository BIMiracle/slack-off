cmake_minimum_required(VERSION 3.20)
project(SlackOff VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use MSVC and the Windows SDK on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(SlackOff WIN32 src/main.cpp src/platform.cpp src/hotkeys.cpp src/settings.cpp resources/app.rc)
target_include_directories(SlackOff PRIVATE resources)
target_compile_definitions(SlackOff PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SlackOff PRIVATE /W4 /utf-8 /permissive- $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
target_link_options(SlackOff PRIVATE /MANIFEST:NO $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
target_link_libraries(SlackOff PRIVATE comctl32 comdlg32 shell32 advapi32 dwmapi)
enable_testing()
add_test(NAME native_integration COMMAND SlackOff --self-test)
